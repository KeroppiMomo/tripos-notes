For any $\psi(\v x)$, $$ \begin{align*} [\hat x_i, \hat x_j] \psi(\v x) ={}& (x_i x_j - x_j x_i) \psi(\v x) = 0, \\ [\hat p_i, \hat p_j] \psi(\v x) ={}& (-i\hbar)^2 \left(\pdd{x_i} \pdd{x_j} - \pdd{x_j} \pdd{x_i}\right)\psi(\v x) = 0 \\ [\hat x_i, \hat p_j] \psi(\v x) ={}& -i\hbar \left(x_i \pdfd{\psi}{x_j} - \pdd{x_j}(x_i \psi)\right) \\ ={}&-i\hbar \left(x_i \pdfd{\psi}{x_j} - x_i \pdfd{\psi}{x_j} - \pdfd{x_i}{x_j}\psi\right) \\ ={}& i\hbar \delta_{ij} \psi \\ [\hat L_i, \hat L_j]\psi ={}& \epsilon_{ijk}\epsilon_{pqk} \hat L_p \hat L_q \psi \\ ={}& -\hbar^2 \epsilon_{ijk} \epsilon_{pqk} \epsilon_{pmn} x_m \pdd{x_n}\left(\epsilon_{qrs} x_r\pdfd{\psi}{x_s}\right) \\ ={}& -\hbar^2 \epsilon_{ijk} \epsilon_{pqk} \epsilon_{pmn} \epsilon_{qrs} (\delta_{rn} x_m \pdfd{\psi}{x_s} + \underbrace{x_m x_r \pdfxd\psi{x_n}{x_s}}_{\text{0 since symmetric}}) \\ ={}& -\hbar^2 \epsilon_{ijk} (\epsilon_{mks} - \epsilon_{nns}\delta_{km}) x_m \pdfd{\psi}{x_s} \\ ={}& i\hbar \epsilon_{ijk} (-i\hbar \epsilon_{kms} x_m \pdfd{\psi}{x_s}) \\ ={}& i\hbar \epsilon_{ijk} \hat L_k\psi \\ [\hat L_i, \hat L^2] ={}& [\hat L_i, \hat L_j \hat L_j] \\ ={}& \hat L_j [\hat L_i, \hat L_j] + [\hat L_i, \hat L_j] \hat L_j \\ ={}& i\hbar \epsilon_{ijk} \underbrace{(\hat L_j \hat L_k + \hat L_k \hat L_j)}_{\text{symmetric}} \\ ={}& 0 \\ [\hat L_i, \hat x_j]\psi ={}& -i\hbar \epsilon_{ikl} \left(x_k \pdd{x_l}(x_j \psi) - x_j x_k\pdd{x_l}\psi\right) \\ ={}& -i\hbar \epsilon_{ikl} x_k \delta_{jl} \psi \\ ={}&i\hbar \epsilon_{ijk} x_k \psi \\ [\hat L_i, \hat p_j] \psi ={}& (-i\hbar)^2 \epsilon_{ikl} \left(x_k \pdfxd{\psi}{x_l}{x_j} - \pdd{x_j}\left(x_k \pdfd{\psi}{x_l}\right)\right) \\ ={}& -(-i\hbar)^2 \epsilon_{ikl} \delta_{jk} \pdfd{\psi}{x_l} \\ ={}& i\hbar \epsilon_{ijk} \hat p_k\psi \end{align*} $$
