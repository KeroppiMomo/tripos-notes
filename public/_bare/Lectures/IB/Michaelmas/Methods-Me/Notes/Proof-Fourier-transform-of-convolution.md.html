$$ \begin{align*} \mathcal{F}_{x\mapsto \lambda}[f* g(x)] ={}& \int e^{-i\lambda x} (f*g)(x) \d x \\ ={}& \int e^{-i\lambda x} \int f(x-y) \, g(y) \d y \d x \\ ={}& \iint e^{-i\lambda (z+y)}\, f(z)\, g(y) \d y \d z \\ ={}& \left(\int e^{-i\lambda z} f(z) \d z\right)\left(\int e^{-i\lambda y} g(y) \d y\right) \\ ={}& \hat f(\lambda) \, \hat g(\lambda). \end{align*} $$
