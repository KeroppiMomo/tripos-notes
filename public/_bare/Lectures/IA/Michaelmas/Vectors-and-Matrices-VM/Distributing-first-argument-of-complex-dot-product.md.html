$$\begin{align*} (\lambda\v u +\mu\v v)\cdot\v w &= \overline{\v w\cdot(\lambda\v u+\mu\v v)} \\ &= \overline{\lambda\v w\cdot\v u + \mu\v w\cdot\v v} \\ &= \overline{\lambda}\,\overline{\v w\cdot\v u} + \overline{\mu}\overline{\v w\cdot\v v} \\ &= \overline\lambda \v u\cdot\v w + \overline\mu\v v\cdot\v w. \end{align*}$$
